async page=>{const rows=[];for(const width of [360,767,768,1000,1001,1440]){await page.setViewportSize({width,height:900});await page.goto(__ORIGIN__+'/app/materials/documents');await page.locator('.material-list li').first().waitFor();for(const row of await page.locator('.material-list li').all()){const data=await row.evaluate(e=>{const children=[...e.children].map(c=>{const r=c.getBoundingClientRect();return{tag:c.tagName,class:c.className,x:r.x,y:r.y,right:r.right,bottom:r.bottom}});return children;});for(let i=0;i<data.length;i++)for(let j=i+1;j<data.length;j++){const a=data[i],b=data[j];if(Math.min(a.right,b.right)-Math.max(a.x,b.x)>1&&Math.min(a.bottom,b.bottom)-Math.max(a.y,b.y)>1)throw Error('overlapping list cells '+JSON.stringify({width,a,b}));}rows.push({width,cells:data.length});}}return{passed:true,rows};}
