import net from'node:net';import{readFileSync,writeFileSync}from'node:fs';import{resolve}from'node:path'
const {port}=JSON.parse(readFileSync('output/breakpoint-zoom-addon/rdp.json'));const socket=net.connect({host:'127.0.0.1',port});let buffer=Buffer.alloc(0),pending=[],queued=[];socket.setTimeout(15000,()=>socket.destroy(Error('RDP timeout')))
function next(){if(queued.length)return Promise.resolve(queued.shift());return new Promise((resolve,reject)=>pending.push({resolve,reject}))}
socket.on('error',e=>{for(const p of pending)p.reject(e);pending=[]});socket.on('data',bytes=>{buffer=Buffer.concat([buffer,bytes]);for(;;){const i=buffer.indexOf(58);if(i<0)return;const n=Number(buffer.subarray(0,i));if(!Number.isInteger(n)||n<0)throw Error('invalid frame');if(buffer.length<i+1+n)return;const msg=JSON.parse(buffer.subarray(i+1,i+1+n));buffer=buffer.subarray(i+1+n);if(pending.length)pending.shift().resolve(msg);else queued.push(msg)}})
function send(message){const b=Buffer.from(JSON.stringify(message));socket.write(Buffer.concat([Buffer.from(b.length+':'),b]))}
try{const hello=await next();if(hello.from!=='root')throw Error('unexpected greeting');send({to:'root',type:'getRoot'});const root=await next();if(!root.addonsActor)throw Error('missing addons actor');send({to:root.addonsActor,type:'installTemporaryAddon',addonPath:resolve('output/breakpoint-zoom-addon'),openDevTools:false});let installed=await next();if(installed.error)throw Error(installed.error+':'+installed.message);if(!installed.addon)throw Error('missing addon result');writeFileSync('output/breakpoint-browser/native-addon-install.json',JSON.stringify({time:new Date().toISOString(),temporary:true,native_API:'browser.tabs.setZoom(tab.id,2)',result:installed},null,2)+'\n');console.log(JSON.stringify({installed:installed.addon.id,temporary:true,native_API:'browser.tabs.setZoom(tab.id,2)'}))}finally{socket.end()}
