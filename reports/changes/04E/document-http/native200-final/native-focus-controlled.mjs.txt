import{readFileSync,writeFileSync}from'node:fs';import{spawnSync,execFileSync}from'node:child_process';import{createHash}from'node:crypto';import assert from'node:assert/strict'
const p='frontend/src/documents/Pages.tsx',fixed=readFileSync(p),old=execFileSync('git',['show','HEAD:'+p]),results=[]
function run(label,args){const r=spawnSync(process.execPath,args,{encoding:'utf8',timeout:120000,maxBuffer:2<<20});writeFileSync('output/document-browser/'+label+'.txt',(r.stdout??'')+(r.stderr??''));results.push({label,exit_code:r.status});return r.status}
try{writeFileSync(p,old);assert.equal(run('native-controlled-old-build',['output/pr57-rebuild-frontend.mjs']),0);assert.equal(run('native-controlled-old-refresh',['output/run-document-browser-code.mjs','output/native-refresh-artifact.code']),0);assert.equal(run('native-controlled-red',['output/run-document-browser-code.mjs','output/native-citation-loop.code']),1,'Reviewed pre-fix focus logic must fail actual native200 regression')}
finally{writeFileSync(p,fixed);assert.equal(run('native-controlled-restored-build',['output/pr57-rebuild-frontend.mjs']),0);assert.equal(run('native-controlled-restored-refresh',['output/run-document-browser-code.mjs','output/native-refresh-artifact.code']),0);assert.equal(run('native-controlled-green',['output/run-document-browser-code.mjs','output/native-citation-loop.code']),0);assert.deepEqual(readFileSync(p),fixed);writeFileSync('output/document-browser/native-controlled.json',JSON.stringify({time:new Date().toISOString(),scenario:'DI-SC-13',mutation:'temporarily restore reviewed pre-fix requestAnimationFrame focus logic',source_sha256:createHash('sha256').update(fixed).digest('hex'),results,restored:true},null,2)+'\n');console.log(JSON.stringify({results,restored:true}))}
